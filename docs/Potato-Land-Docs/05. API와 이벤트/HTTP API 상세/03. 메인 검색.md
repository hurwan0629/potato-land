# HTTP API 상세 - 메인/검색

## 4. 메인/검색

### GET `/api/main`

Request:

```json
{
  "cookies": {
    "access_token": "optional"
  },
  "query": {
    "limit": 8
  }
}
```

Success 200:

```json
{
  "success": true,
  "data": {
    "usedListings": [
      {
        "listingIdx": 1,
        "listingType": "USED",
        "title": "감자 키보드",
        "thumbnailUrl": "/resources/listings/1-1.png",
        "price": 10000,
        "tradeStatus": "ON_SALE",
        "sellerNickname": "판매자",
        "createdAt": "2026-07-30T13:00:00.000Z"
      }
    ],
    "auctionListings": [
      {
        "listingIdx": 2,
        "listingType": "AUCTION",
        "title": "경매 상품",
        "thumbnailUrl": "/resources/listings/2-1.png",
        "currentPrice": 20000,
        "status": "ON_GOING",
        "endsAt": "2026-07-31T13:00:00.000Z"
      }
    ],
    "popularListings": []
  }
}
```

Exceptions:

```json
[
  {
    "http": 400,
    "code": "VALIDATION_ERROR",
    "message": "limit 값이 올바르지 않습니다.",
    "details": { "field": "limit" }
  }
]
```

### GET `/api/categories`

Request:

```json
{
  "cookies": {},
  "query": {}
}
```

Success 200:

```json
{
  "success": true,
  "data": {
    "items": [
      {
        "categoryIdx": 1,
        "name": "디지털",
        "sortOrder": 1
      }
    ]
  }
}
```

Exceptions:

```json
[]
```

### GET `/api/search`

Request:

```json
{
  "cookies": {
    "access_token": "optional"
  },
  "query": {
    "q": "키보드",
    "type": "ALL",
    "categoryIdx": 1,
    "minPrice": 1000,
    "maxPrice": 100000,
    "page": 1,
    "limit": 10
  }
}
```

Success 200:

```json
{
  "success": true,
  "data": {
    "items": [
      {
        "listingIdx": 1,
        "listingType": "USED",
        "title": "감자 키보드",
        "thumbnailUrl": "/resources/listings/1-1.png",
        "price": 10000,
        "status": "ON_SALE",
        "sellerNickname": "판매자",
        "createdAt": "2026-07-30T13:00:00.000Z"
      }
    ],
    "page": 1,
    "limit": 10,
    "totalCount": 1,
    "totalPages": 1
  }
}
```

Exceptions:

```json
[
  {
    "http": 400,
    "code": "VALIDATION_ERROR",
    "message": "검색 조건이 올바르지 않습니다.",
    "details": { "fields": ["type", "categoryIdx", "minPrice", "maxPrice", "page", "limit"] }
  }
]
```
